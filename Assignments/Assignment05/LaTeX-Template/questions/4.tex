\subsectionaddtolist{الف}

با توجه به این که سیستم تمام‌pass (All-pass) است، اندازه پاسخ فرکانسی آن برای تمام فرکانس‌ها برابر با ۱ است ($|H(e^{j\omega})| = 1$). در سیستم‌های تمام‌pass حقیقی، به ازای هر قطب در نقطه $p_i$، یک صفر در نقطه معکوس و مزدوج آن یعنی $z_i = \frac{1}{p_i^*}$ قرار دارد.

\begin{enumerate}
    \item \textbf{مکان قطب‌ها:} سیستم دارای دو قطب مختلط در شعاع $r = 0.8$ و زوایای $\theta = \pm \frac{\pi}{4}$ است:
    \begin{equation*}
    p_1 = 0.8 e^{j \frac{\pi}{4}}, \quad p_2 = 0.8 e^{-j \frac{\pi}{4}}
    \end{equation*}
    عامل مخرج تابع تبدیل (قطب‌ها) برابر است با:
    \begin{equation*}
    D(z) = (1 - p_1 z^{-1})(1 - p_2 z^{-1}) = 1 - 2(0.8)\cos\left(\frac{\pi}{4}\right)z^{-1} + (0.8)^2 z^{-2}
    \end{equation*}
    \begin{equation*}
    D(z) = 1 - 1.6 \left(\frac{\sqrt{2}}{2}\right)z^{-1} + 0.64 z^{-2} = 1 - 0.8\sqrt{2}z^{-1} + 0.64 z^{-2}
    \end{equation*}

    \item \textbf{مکان صفرها:} صفرها در مکان‌های معکوس و مزدوج قطب‌ها قرار دارند:
    \begin{equation*}
    z_1 = \frac{1}{p_1^*} = \frac{1}{0.8 e^{-j \frac{\pi}{4}}} = 1.25 e^{j \frac{\pi}{4}}, \quad z_2 = \frac{1}{p_2^*} = \frac{1}{0.8 e^{j \frac{\pi}{4}}} = 1.25 e^{-j \frac{\pi}{4}}
    \end{equation*}
    بنابراین صفرها دقیقاً در شعاع $r = 1.25$ و زوایای $\pm \frac{\pi}{4}$ قرار دارند. عامل صورت تابع تبدیل (صفرها) با ساختار تمام‌pass به شکل زیر است:
    \begin{equation*}
    N(z) = z^{-2} D(z^{-1}) = 0.64 - 0.8\sqrt{2}z^{-1} + z^{-2}
    \end{equation*}
\end{enumerate}

برای این که بهره در فرکانس فرضی یک شود، شکل استاندارد تمام‌pass مرتبه دو را به صورت زیر می‌نویسیم:
\begin{equation*}
H(z) = \frac{0.64 - 0.8\sqrt{2}z^{-1} + z^{-2}}{1 - 0.8\sqrt{2}z^{-1} + 0.64 z^{-2}}
\end{equation*}

\textbf{تعیین ناحیه همگرایی (ROC):}
چون در صورت سوال قید شده است که سیستم \textbf{پایدار} است، ناحیه همگرایی حتماً باید شامل دایره واحد ($|z| = 1$) باشد. از طرفی چون سیستم با توجه به موقعیت قطب‌ها درون دایره واحد می‌تواند علی باشد، ناحیه همگرایی آن به صورت بیرونی و فراتر از بزرگترین قطب ($r = 0.8$) تعریف می‌شود:
\begin{equation*}
\text{ROC: } |z| > 0.8
\end{equation*}

\subsectionaddtolist{ب}

می‌خواهیم نشان دهیم که روی دایره واحد ($z = e^{j\omega}$)، اندازه تابع تبدیل دقیقاً برابر با ۱ است. 
تابع تبدیل را به فرم ضرب عوامل بازنویسی می‌کنیم:
\begin{equation*}
H(z) = \frac{(z^{-1} - p_1^*)(z^{-1} - p_2^*)}{(1 - p_1 z^{-1})(1 - p_2 z^{-1})} = z^{-2} \frac{(1 - p_1^* z)(1 - p_2^* z)}{(1 - p_1 z^{-1})(1 - p_2 z^{-1})}
\end{equation*}

با جایگذاری $z = e^{j\omega}$ روی دایره واحد، توجه می‌کنیم که $z^{-1} = e^{-j\omega} = z^*$. بنابر این داریم:
\begin{equation*}
|H(e^{j\omega})| = |e^{-j2\omega}| \cdot \frac{|(e^{-j\omega} - p_1^*)(e^{-j\omega} - p_2^*)|}{|(1 - p_1 e^{-j\omega})(1 - p_2 e^{-j\omega})|}
\end{equation*}
می‌دانیم $|e^{-j2\omega}| = 1$. همچنین برای تک‌تک جملات صورت و مخرج، اگر از عامل داخل قدرمطلق فاکتور بگیریم:
\begin{equation*}
|e^{-j\omega} - p_i^*| = |e^{-j\omega}(1 - p_i^* e^{j\omega})| = |e^{-j\omega}| \cdot |1 - p_i^* e^{j\omega}| = |1 - p_i^* e^{j\omega}|
\end{equation*}
چون اندازه هر عدد مختلط با اندازه مزدوج آن برابر است ($|W| = |W^*|$):
\begin{equation*}
|1 - p_i^* e^{j\omega}| = |(1 - p_i^* e^{j\omega})^*| = |1 - p_i e^{-j\omega}|
\end{equation*}

بنابراین، صورت و مخرج کسر در رابطه $|H(e^{j\omega})|$ نظیر به نظیر با هم ساده می‌شوند:
\begin{equation*}
|H(e^{j\omega})| = 1 \times \frac{|1 - p_1 e^{-j\omega}| \cdot |1 - p_2 e^{-j\omega}|}{|1 - p_1 e^{-j\omega}| \cdot |1 - p_2 e^{-j\omega}|} = 1
\end{equation*}
این اثبات نشان می‌دهد چیدمان جفت قطب و صفر متناظر معکوس همواره اندازه‌ای برابر با یک تولید می‌کند.

\subsectionaddtolist{ج}

یک سیستم زمانی حداقل فاز (Minimum-Phase) است که هم قطب‌ها و هم صفرهای آن در داخل دایره واحد ($|z| < 1$) قرار داشته باشند. سیستم حداقل فاز متناظر با حفظ اندازه پاسخ فرکانسی، صفرهای خارج دایره واحد ($1.25 e^{\pm j\frac{\pi}{4}}$) را به داخل منتقل می‌کند (یعنی معکوس و مزدوج آن‌ها را می‌گیرد):
\begin{equation*}
z_{min} = \frac{1}{(1.25 e^{\pm j\frac{\pi}{4}})^*} = 0.8 e^{\pm j\frac{\pi}{4}}
\end{equation*}

برای این که اندازه پاسخ فرکانسی دقیقاً حفظ شود ($|H_{min}(e^{j\omega})| = |H(e^{j\omega})| = 1$)، صفرهای جدید دقیقاً بر روی قطب‌های سیستم اصلی منطبق شده و آن‌ها را خنثی می‌کنند. بهره نهایی برابر با ۱ شده و تابع تبدیل سیستم حداقل فاز به صورت زیر به دست می‌آید:
\begin{equation*}
H_{min}(z) = 1
\end{equation*}

\textbf{محاسبه نمونه اول با استفاده از قضیه مقدار اولیه (IVT):}
قضیه مقدار اولیه بیان می‌کند که $h[0] = \lim_{z \to \infty} H(z)$.

۱. برای سیستم اصلی $H(z)$:
\begin{equation*}
h[0] = \lim_{z \to \infty} \frac{0.64 - 0.8\sqrt{2}z^{-1} + z^{-2}}{1 - 0.8\sqrt{2}z^{-1} + 0.64 z^{-2}} = \frac{0.64}{1} = 0.64
\end{equation*}

۲. برای سیستم حداقل فاز $H_{min}(z)$:
\begin{equation*}
h_{min}[0] = \lim_{z \to \infty} (1) = 1
\end{equation*}

\textbf{بررسی ویژگی تراکم انرژی (Energy Concentration):}
خاصیت سیستم‌های حداقل فاز این است که در میان تمامی سیستم‌هایی که اندازه پاسخ فرکانسی یکسانی دارند، سیستم حداقل فاز بیشترین تمرکز انرژی را در نمونه‌های اولیه خود (نزدیک به مبدأ زمان $n=0$) دارد. 
در این حالت خاص مشاهده می‌کنیم که:
\begin{equation*}
|h_{min}[0]| = 1 > |h[0]| = 0.64
\end{equation*}
این نتیجه دقیقاً تاییدکننده ویژگی تراکم انرژی در سیستم حداقل فاز است؛ چرا که سیستم حداقل فاز تمام انرژی خود را در همان نمونه اول ($n=0$) آزاد کرده است ($h_{min}[n] = \delta[n]$)، در حالی که در سیستم غیر حداقل فاز اصلی، بخشی از انرژی به زمان‌های جلوتر منتقل شده است.
