\subsectionaddtolist{الف}

\setLTR
$
input \in R^{3} \\
output_1 \in R^{4} \ \ \ output_2 \in R^{2} \ \ \ output_3 \in R^{1} \\
w_1 \in R^{4\times 3} \ \ \ w_2 \in R^{2\times 4} \ \ \ w_3 \in R^{1 \times 2} \\
b_1 \in R^{4\times 1} \ \ \ \ b_2 \in R^{2\times 1} \ \ \ \ b_3 \in R^{1 \times1} 
$
\setRTL

\subsectionaddtolist{ب}
\setLTR
$
x \in R^{3\times m}
$
\setRTL
\begin{itemize}
    \item{لایه اول}
    \setLTR
    $\\
    z_1 = w_1 x + b_1 \in R^{4\times m} \\
    a_1 = Relu(z_1) \in R^{4\times m} \\
    $
    \setRTL
    \item{لایه دوم}
    \setLTR
    $\\
    z_2 = w_2 a_1 + b_2 \in R^{2\times m} \\
    a_2 = \tanh(z_2) \in R^{2\times m} \\
    $
    \setRTL
    \item{لایه خروجی}
    \setLTR
    $\\
    z_3 = w_3 a_2 + b_3 \in R^{1\times m} \\
    \hat{y} = \sigma(z_3) \in R^{1\times m} \\
    $
    \setRTL
\end{itemize}


\subsectionaddtolist{ج}
فرمول تابع هزینه به صورت زیر است:
\[
J = \frac{-1}{m} \sum_{i=1}^{m} \left( y_i \log(\hat{y}_i) + (1-y_i) \log(1-\hat{y}_i) \right)
\]
با توجه به قاعده زنجیره، مشتق تابع هزینه نسبت به وزن لایه اول به صورت زیر محاسبه می‌شود:
\[
\frac{\partial J}{\partial w_1} = \frac{\partial J}{\partial z_3} \cdot \frac{\partial z_3}{\partial a_2} \cdot \frac{\partial a_2}{\partial z_2} \cdot \frac{\partial z_2}{\partial a_1} \cdot \frac{\partial a_1}{\partial z_1} \cdot \frac{\partial z_1}{\partial w_1}
\]

\setLTR
$
\frac{\partial J}{\partial z_3} = \frac{1}{m} (\hat{y} - y) \in R^{1\times m} \\
\frac{\partial J}{\partial w_3} = \frac{1}{m} (\hat{y} - y) \cdot a_2^T \in R^{1\times 2} \\
\\
\frac{\partial J}{\partial A_2} = w_3^T \cdot \frac{1}{m} (\hat{y} - y) \in R^{2\times m} \\
\frac{\partial J}{\partial z_2} = \frac{\partial J}{\partial A_2} \odot (1 - \tan^2(z_2)) \in R^{2\times m} \\
\frac{\partial J}{\partial w_2} = \frac{1}{m} \left( \frac{\partial J}{\partial z_2} \cdot a_1^T \right) \in R^{2\times 4} \\
\\
\frac{\partial J}{\partial A_1} = w_2^T \cdot \frac{\partial J}{\partial z_2} \in R^{4\times m} \\
\frac{\partial J}{\partial z_1} = \frac{\partial J}{\partial A_1} \odot (\frac{d}{dz}Relu(z_1)) \in R^{4\times m} \\
\frac{\partial J}{\partial w_1} = \frac{1}{m}\frac{\partial J}{\partial z_1}\frac{}{} x^T \in R^{4\times 3}\\
$
\setRTL
\pagebreak

\subsectionaddtolist{د}
\textbf{محاسبه \lr{Forward Pass}:}

\setLTR
$
x = \begin{bmatrix} 1 \\ 2 \\ 3 \end{bmatrix}, \quad m = 1
$

$
z_1 = w_1 x + b_1 = 
\begin{bmatrix}
1 & 0 & -1 \\
2 & -1 & 0 \\
0 & 1 & 1 \\
-1 & -2 & 1 
\end{bmatrix}
\begin{bmatrix}
1 \\ 2 \\ 3 
\end{bmatrix}
+ 
\begin{bmatrix}
0 \\ 1 \\ -1 \\ 0 
\end{bmatrix}
=
\begin{bmatrix}
1(1) + 0(2) + (-1)(3) \\
2(1) -1(2) + 0(3) \\
0(1) + 1(2) + 1(3) \\
-1(1) -2(2) + 1(3)
\end{bmatrix}
+
\begin{bmatrix}
0 \\ 1 \\ -1 \\ 0
\end{bmatrix}
=
\begin{bmatrix}
-2 \\ 1 \\ 4 \\ -2
\end{bmatrix}
$

$
a_1 = \text{ReLU}(z_1) = 
\begin{bmatrix}
0 \\ 1 \\ 4 \\ 0
\end{bmatrix}
$

$
z_2 = w_2 a_1 + b_2 = 
\begin{bmatrix}
1 & -1 & 0 & 2 \\
0 & 1 & -1 & -1 
\end{bmatrix}
\begin{bmatrix}
0 \\ 1 \\ 4 \\ 0
\end{bmatrix}
=
\begin{bmatrix}
(1)(0) + (-1)(1) + (0)(4) + (2)(0) \\
(0)(0) + (1)(1) + (-1)(4) + (-1)(0)
\end{bmatrix}
=
\begin{bmatrix}
-1 \\ -3
\end{bmatrix}
$

$
a_2 = \tanh(z_2) = \begin{bmatrix}
\tanh(-1) \\ \tanh(-3)
\end{bmatrix} \approx \begin{bmatrix}
-0.76 \\ -0.995
\end{bmatrix}
$

$
z_3 = w_3 a_2 + b_3 = 
\begin{bmatrix} 1 & -1 \end{bmatrix}
\begin{bmatrix}
-0.76 \\ -0.995
\end{bmatrix}
= (-0.76) - (-0.995) = 0.235
$

$
\hat{y} = \sigma(z_3) = \frac{1}{1 + e^{-0.235}} \approx 0.5585
$
\setRTL

\textbf{محاسبه \lr{Backward Pass}:}

\setLTR
$
\frac{\partial J}{\partial z_3} = \hat{y} - y = 0.5585 - 1 = -0.4415
$

$
\frac{\partial J}{\partial w_3} = \frac{1}{m} \cdot (\hat{y} - y) \cdot a_2^T = (-0.4415) \cdot \begin{bmatrix} -0.76 & -0.995 \end{bmatrix}
= \begin{bmatrix} 0.3355 & 0.4393 \end{bmatrix}
$

$
\frac{\partial J}{\partial b_3} = \frac{\partial J}{\partial z_3} = -0.4415
$

$
\frac{\partial J}{\partial a_2} = w_3^T \cdot \frac{\partial J}{\partial z_3} = 
\begin{bmatrix} 1 \\ -1 \end{bmatrix} \cdot (-0.4415)
= \begin{bmatrix} -0.4415 \\ 0.4415 \end{bmatrix}
$

$
\frac{\partial J}{\partial z_2} = \frac{\partial J}{\partial a_2} \odot (1 - \tanh^2(z_2)) \approx 
\begin{bmatrix} -0.4415 \\ 0.4415 \end{bmatrix} 
\odot \begin{bmatrix} 1 - 0.5776 \\ 1 - 0.9900 \end{bmatrix}
= \begin{bmatrix} -0.4415 \cdot 0.4224 \\ 0.4415 \cdot 0.01 \end{bmatrix}
= \begin{bmatrix} -0.1865 \\ 0.0044 \end{bmatrix}
$

$
\frac{\partial J}{\partial w_2} = \frac{\partial J}{\partial z_2} \cdot a_1^T =
\begin{bmatrix}
-0.1865 \\
0.0044
\end{bmatrix}
\begin{bmatrix}
0 & 1 & 4 & 0
\end{bmatrix}
=
\begin{bmatrix}
0 & -0.1865 & -0.746 & 0 \\
0 & 0.0044 & 0.0176 & 0
\end{bmatrix}
$

$
\frac{\partial J}{\partial b_2} = \frac{\partial J}{\partial z_2} = \begin{bmatrix} -0.1865 \\ 0.0044 \end{bmatrix}
$

$
\frac{\partial J}{\partial a_1} = w_2^T \cdot \frac{\partial J}{\partial z_2} =
\begin{bmatrix}
1 & 0 \\
-1 & 1 \\
0 & -1 \\
2 & -1 
\end{bmatrix}
\cdot
\begin{bmatrix}
-0.1865 \\ 0.0044
\end{bmatrix}
=
\begin{bmatrix}
-0.1865 \\
0.1909 \\
-0.0044 \\
-0.3774
\end{bmatrix}
$

$
\frac{\partial J}{\partial z_1} = \frac{\partial J}{\partial a_1} \odot \text{ReLU}'(z_1) =
\begin{bmatrix}
0 \\ 0.1909 \\ -0.0044 \\ 0
\end{bmatrix}
$

$
\frac{\partial J}{\partial w_1} = \frac{\partial J}{\partial z_1} \cdot x^T =
\begin{bmatrix}
0 \\ 0.1909 \\ -0.0044 \\ 0
\end{bmatrix}
\begin{bmatrix}
1 & 2 & 3
\end{bmatrix}
=
\begin{bmatrix}
0 & 0 & 0 \\
0.1909 & 0.3818 & 0.5727 \\
-0.0044 & -0.0088 & -0.0132 \\
0 & 0 & 0
\end{bmatrix}
$

$
\frac{\partial J}{\partial b_1} = \frac{\partial J}{\partial z_1} =
\begin{bmatrix}
0 \\ 0.1909 \\ -0.0044 \\ 0
\end{bmatrix}
$
\setRTL
